#include "pub.hpp"
#include "sub.hpp"
#include <gurumdds/cpp/core/Exception.hpp>
#include <gurumdds/cpp/domain/ddsdomain.hpp>
#include <exception>
#include <iostream>
#include <string>
namespace {
inline constexpr dds::domain::DomainId DOMAIN_ID = 0;
inline constexpr char TOPIC_NAME[] = "HelloWorld";
} // namespace
int main(int argc, char* argv[]) {
try {
if(argc == 2) {
const std::string mode = argv[1];
if(mode == "pub") {
helloworld_deadline::Publisher pub(DOMAIN_ID, TOPIC_NAME);
pub.Run();
return 0;
} else if(mode == "sub") {
helloworld_deadline::Subscriber sub(DOMAIN_ID, TOPIC_NAME);
sub.Run();
return 0;
}
}
} catch(const dds::core::Exception& error) {
std::cerr << "DDS error: " << error.what() << '\n';
return -1;
} catch(const std::exception& error) {
std::cerr << "Error: " << error.what() << '\n';
return -1;
}
std::cerr << "Usage: helloworld_deadline <pub|sub>\n";
return -1;
}

Subscriber::Subscriber(dds::domain::DomainId domainId, const std::string& topicName)
: m_participant(domainId),
m_topic(m_participant, topicName),
m_subscriber(m_participant, m_participant.default_subscriber_qos()
<< dds::core::policy::EntityFactory::ManuallyEnable()),
// Allow more than the publisher's nominal sample interval to account for network and scheduling delays.
m_reader(m_subscriber, m_topic,
m_subscriber.default_datareader_qos() << dds::core::policy::Deadline(dds::core::Duration(1, 0))) {}
void Subscriber::Run() {
using dds::core::status::StatusMask;
auto mask = StatusMask::requested_deadline_missed();
mask |= StatusMask::data_available();
m_reader.listener(&m_listener, mask);
std::cout << FormatTimestamp(std::chrono::duration_cast<std::chrono::nanoseconds>(
std::chrono::system_clock::now().time_since_epoch())
.count())
<< " [SUB] Deadline: 1 s | Waiting for HelloWorld." << std::endl;
m_reader.enable();
while(true) {
std::this_thread::sleep_for(std::chrono::seconds(1));
}
}

Subscriber::Listener::on_requested_deadline_missed(dds::sub::DataReader<HelloWorld>&,

void Subscriber::Listener::on_data_available(dds::sub::DataReader<HelloWorld>& reader) {
for(const auto& sample: reader.take()) {
if(!sample.info().valid()) {
continue;
}
const auto& data = sample.data();
const std::lock_guard lock(g_outputMutex);
std::cout << FormatTimestamp(std::chrono::duration_cast<std::chrono::nanoseconds>(
std::chrono::system_clock::now().time_since_epoch())
.count())
<< " [SUB] Received HelloWorld | user_id=" << data.user_id << " | message=\""
<< (data.message != nullptr ? data.message : "null") << '"' << std::endl;
}
}
} // namespace helloworld_deadline

#ifndef HELLOWORLD_DEADLINE_TIMESTAMP_HPP
#define HELLOWORLD_DEADLINE_TIMESTAMP_HPP
#include <chrono>
#include <cstdint>
#include <ctime>
#include <iomanip>
#include <sstream>
#include <string>
namespace helloworld_deadline {
inline std::string FormatTimestamp(std::int64_t timestamp) {
const auto elapsed = std::chrono::floor<std::chrono::milliseconds>(std::chrono::nanoseconds(timestamp));
const auto seconds = std::chrono::floor<std::chrono::seconds>(elapsed);
const auto milliseconds = elapsed - seconds;
const auto time = static_cast<std::time_t>(seconds.count());
std::tm localTime{};
#ifdef _WIN32
if(localtime_s(&localTime, &time) != 0) {
#else
if(localtime_r(&time, &localTime) == nullptr) {
#endif
return "unavailable";
}
std::ostringstream output;
output << std::put_time(&localTime, "%Y-%m-%d %H:%M:%S") << '.' << std::setfill('0') << std::setw(3)
<< milliseconds.count();
return output.str();
}
} // namespace helloworld_deadline
#endif // HELLOWORLD_DEADLINE_TIMESTAMP_HPP

cmake_minimum_required(VERSION 3.20)
project(helloworld_deadline VERSION 1.0.0 LANGUAGES CXX)
find_package(Threads REQUIRED)
set(HELLOWORLD_DEADLINE_IDL "${CMAKE_CURRENT_SOURCE_DIR}/HelloWorld.idl")
set(HELLOWORLD_DEADLINE_GENERATED_DIR "${CMAKE_CURRENT_SOURCE_DIR}/generated")
set(HELLOWORLD_DEADLINE_TYPES_SRCS "${HELLOWORLD_DEADLINE_GENERATED_DIR}/HelloWorldTypeSupport.cpp")
set(HELLOWORLD_DEADLINE_TYPES_HEADERS
"${HELLOWORLD_DEADLINE_GENERATED_DIR}/HelloWorld.hpp"
"${HELLOWORLD_DEADLINE_GENERATED_DIR}/HelloWorldTypeSupport.hpp"
)
add_custom_command(
OUTPUT "${HELLOWORLD_DEADLINE_TYPES_SRCS}" ${HELLOWORLD_DEADLINE_TYPES_HEADERS}
COMMAND "${CMAKE_COMMAND}" -E make_directory "${HELLOWORLD_DEADLINE_GENERATED_DIR}"
COMMAND "${GURUMIDL}" cpp "${HELLOWORLD_DEADLINE_IDL}" "${HELLOWORLD_DEADLINE_GENERATED_DIR}" --replace
DEPENDS "${GURUMIDL}" "${HELLOWORLD_DEADLINE_IDL}"
COMMENT "Generating HelloWorld C++ types from IDL"
VERBATIM
)
add_custom_target(helloworld_deadline_generated
DEPENDS "${HELLOWORLD_DEADLINE_TYPES_SRCS}" ${HELLOWORLD_DEADLINE_TYPES_HEADERS}
)
add_executable(helloworld_deadline
${CMAKE_CURRENT_SOURCE_DIR}/src/main.cpp
${CMAKE_CURRENT_SOURCE_DIR}/src/pub.cpp
${CMAKE_CURRENT_SOURCE_DIR}/src/sub.cpp
"${HELLOWORLD_DEADLINE_TYPES_SRCS}")
add_dependencies(helloworld_deadline helloworld_deadline_generated)
target_include_directories(helloworld_deadline PRIVATE "${HELLOWORLD_DEADLINE_GENERATED_DIR}")
target_link_libraries(helloworld_deadline PRIVATE gurumdds::cxx Threads::Threads)
set_target_properties(helloworld_deadline PROPERTIES OUTPUT_NAME "helloworld_deadline")
