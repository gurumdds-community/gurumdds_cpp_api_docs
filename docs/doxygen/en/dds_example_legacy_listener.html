Listener-based Subscriber

Registers a data-available listener, takes samples in the callback, and processes samples with valid payloads

#include <chrono>
#include <thread>
#include <gurumdds/dds.hpp>
#include <HelloWorldTypeSupport.hpp>
static bool received = false;
class DRListener : public dds::sub::NoOpDataReaderListener<Hello::Msg> {
void on_data_available(dds::sub::DataReader<T>& reader) {
auto samples = reader.take();
for(auto sample : samples) {
if(!sample.info().valid())
continue;
std::cout << "Data received!" << std::endl;
std::cout << "userId: " << sample.data().userId << std::endl;
std::cout << "message: " << sample.data().message << std::endl;
received = true;
}
}
};
int main(int argc, char** argv) {
try {
dds::domain::DomainId domain_id = 0;
dds::domain::DomainParticipant dp(domain_id);
dds::topic::Topic<T> topic(dp, "Hello::Msg");
dds::sub::Subscriber sub(dp);
dds::sub::DataReader<T> dr(sub, topic);
dds::sub::AnyDataReader reader{dr};
auto dr2 = reader.get<Hello::Msg>();
DRListener dr_listener;
dr.listener(&dr_listener, dds::core::status::StatusMask::data_available());
while(!received)
std::this_thread::sleep_for(std::chrono::milliseconds(100));
std::this_thread::sleep_for(std::chrono::milliseconds(1000));
} catch(dds::core::Error e) {
std::cout << e.what();
}
return 0;
}
