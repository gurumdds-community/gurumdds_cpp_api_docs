Hello World Subscriber

데이터 타입을 지정한 DataReader를 만들고 sample을 폴링합니다.

애플리케이션 코드에서 payload를 읽기 전에 SampleInfo::valid()를 확인하세요.

#include <iostream>
#include <ctime>
#include <gurumdds/dds.hpp>
#define MY_MAX_LEN 8
// Generated Code ==============================================================
// Contains Type Structure, CDR Meta information, and CXX topic traits
// GurumIDL MUST be able to provide following codes
struct HelloWorldData_Msg {
dds_Long userId;
dds_String message;
};
#define METADATA_STR_ARR_LEN 3
static const char* metadata_str_arr[METADATA_STR_ARR_LEN] = {
"!1{(type=HelloWorldData::Msg,member=2)",
"i(name=userId)",
"'(name=message)",
};
static dds_TypeSupport* ts;
dds_TypeSupport* HelloWorldData_MsgTypeSupport_get_instance(void) {
if(ts == NULL)
ts = dds_TypeSupport_create2(metadata_str_arr, METADATA_STR_ARR_LEN);
return ts;
}
namespace dds { namespace topic {
template <>
struct is_topic_type<HelloWorldData_Msg> {
enum { value = 1 };
};
template <>
struct topic_type_support<HelloWorldData_Msg> {
static dds_TypeSupport* ref() { return HelloWorldData_MsgTypeSupport_get_instance(); }
};
template <>
struct topic_type_name<HelloWorldData_Msg> {
static std::string value() { return "HelloWorldData_Msg"; }
};
} }
int main(int argc, char** argv) {
try {
dds::domain::DomainId id = 0;
dds::domain::DomainParticipant dp(id);
dds::sub::Subscriber sub(dp);
dds::topic::Topic<T> topic(dp, "HelloWorld");
dds::sub::DataReader<T> dr(sub, topic);
do {
dds::sub::LoanedSamples<HelloWorldData_Msg> samples = dr.take();
if(samples.length() == 0) {
struct timespec delay = { 0, 500 * 1000 * 1000 };
nanosleep(&delay, NULL);
continue;
}
for(auto item : samples) {
std::cout << "Received: " << std::endl;
std::cout << "userId : " << item.data().userId << std::endl;
std::cout << "message: " << item.data().message << std::endl;
}
break;
} while(1);
} catch (const dds::core::Exception& e) {
std::cerr << e.what();
}
return 0;
}
