ContentFilteredTopic을 사용하는 Subscriber

userId>5 식을 적용한 ContentFilteredTopic에 DataReader를 만듭니다.

#include <chrono>
#include <thread>
#include <gurumdds/dds.hpp>
#include <HelloWorldTypeSupport.hpp>
int main(int argc, char** argv) {
try {
dds::domain::DomainId domain_id = 0;
dds::domain::DomainParticipant dp(domain_id);
dds::topic::Topic<T> topic(dp, "HelloWorld");
dds::topic::Filter filter("userId>5");
dds::topic::ContentFilteredTopic<T> cftopic(topic, "HelloWorld_cft", filter);
dds::sub::Subscriber sub(dp);
dds::sub::DataReader<T> dr(sub, cftopic);
bool loop = true;
do {
dds::sub::LoanedSamples<Hello::Msg> samples = dr.take();
if(samples.length() == 0) {
struct timespec delay = { 0, 500 * 1000 * 1000 };
nanosleep(&delay, NULL);
continue;
}
for(auto item : samples) {
if(!item.info().valid())
continue;
std::cout << "Received: " << std::endl;
std::cout << "userId : " << item.data().userId << std::endl;
std::cout << "message: " << item.data().message << std::endl;
if(item.data().userId == 9)
loop = false;
}
} while(loop);
} catch(dds::core::Error e) {
std::cout << e.what();
}
return 0;
}
