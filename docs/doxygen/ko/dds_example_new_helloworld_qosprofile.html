XML에서 QoS profile 읽기

XML에서 QoS profile을 읽고 각 정책을 기본 QoS 값과 비교합니다.

XML QoS profile 읽기와 확인

예제 실행하기

./helloworld_qosprofile

프로그램은 C XML QoS API로 profile을 읽습니다. 먼저 Default QoS -> XML profile QoS를 비교하세요. Base profile -> Inherited profile에서는 profile 상속으로 달라진 값을 확인합니다.

#include "qos_output.hpp"
#include <gurumdds/cpp/core/Exception.hpp>
#include <gurumdds/cpp/domain/ddsdomain.hpp>
#include <gurumdds/cpp/pub/Publisher.hpp>
#include <gurumdds/cpp/sub/Subscriber.hpp>
#include <gurumdds/xml.h>
#include <iostream>
#include <memory>
#include <string_view>
namespace {
inline constexpr char DEFAULT_XML_PATH[] = HELLOWORLD_QOSPROFILE_XML;
inline constexpr char DEFAULT_PROFILE[] = "HelloWorldQos::HelloWorldQosProfile";
inline constexpr char DEFAULT_INHERITED_PROFILE[] = "HelloWorldQos::HelloWorldInheritedQosProfile";
inline constexpr dds::domain::DomainId DOMAIN_ID = 0;
void Run(const char* pXmlPath, const char* pProfileName, const char* pInheritedProfileName) {
auto* pFactory = dds_DomainParticipantFactory_get_instance();
if(!pFactory) {
std::cerr << "[DDS ERROR] Could not initialize the DomainParticipantFactory.\n";
return;
}
// The XML loader modifies existing QoS, including the sequences for list-valued policies.
dds_DataWriterQos xmlWriterQos{};
if(dds_DataWriterQos_copy(&xmlWriterQos, &dds_DATAWRITER_QOS_DEFAULT) != dds_RETCODE_OK) {
std::cerr << "[DDS ERROR] Could not initialize DataWriter QoS for the XML loader.\n";
return;
}
const std::unique_ptr<dds_DataWriterQos, decltype(&dds_DataWriterQos_finalize)> xmlWriterGuard(
&xmlWriterQos, dds_DataWriterQos_finalize);
dds_DataReaderQos xmlReaderQos{};
if(dds_DataReaderQos_copy(&xmlReaderQos, &dds_DATAREADER_QOS_DEFAULT) != dds_RETCODE_OK) {
std::cerr << "[DDS ERROR] Could not initialize DataReader QoS for the XML loader.\n";
return;
}
const std::unique_ptr<dds_DataReaderQos, decltype(&dds_DataReaderQos_finalize)> xmlReaderGuard(
&xmlReaderQos, dds_DataReaderQos_finalize);
const auto writerResult =
dds_xml_QosProfile_get_datawriter_qos(pFactory, pXmlPath, pProfileName, &xmlWriterQos);
if(writerResult != dds_RETCODE_OK) {
std::cerr << "[DDS ERROR] Could not load DataWriter QoS (code " << writerResult << ") for " << pProfileName
<< " from " << pXmlPath << '\n';
return;
}
const auto readerResult =
dds_xml_QosProfile_get_datareader_qos(pFactory, pXmlPath, pProfileName, &xmlReaderQos);
if(readerResult != dds_RETCODE_OK) {
std::cerr << "[DDS ERROR] Could not load DataReader QoS (code " << readerResult << ") for " << pProfileName
<< " from " << pXmlPath << '\n';
return;
}
// The C++ QoS objects share the sequences owned by the loaded C profiles.
const dds::pub::qos::DataWriterQos profileWriterQos(xmlWriterQos);
const dds::sub::qos::DataReaderQos profileReaderQos(xmlReaderQos);
const dds::domain::DomainParticipant participant(DOMAIN_ID);
const dds::pub::Publisher publisher(participant);
const dds::sub::Subscriber subscriber(participant);
const auto defaultWriterQos = publisher.default_writer_qos();
const auto defaultReaderQos = subscriber.default_datareader_qos();
std::cout << "XML: " << pXmlPath << "\nProfile: " << pProfileName << "\nDefault QoS -> XML profile QoS\n";
helloworld_qosprofile::PrintWriterQos(std::cout, defaultWriterQos, profileWriterQos);
helloworld_qosprofile::PrintReaderQos(std::cout, defaultReaderQos, profileReaderQos);
if(!pInheritedProfileName) {
return;
}
// Start with the same defaults so GurumDDS resolves the inheritance in the XML itself.
dds_DataWriterQos inheritedWriterQos{};
if(dds_DataWriterQos_copy(&inheritedWriterQos, &dds_DATAWRITER_QOS_DEFAULT) != dds_RETCODE_OK) {
std::cerr << "[DDS ERROR] Could not initialize inherited DataWriter QoS.\n";
return;
}
const std::unique_ptr<dds_DataWriterQos, decltype(&dds_DataWriterQos_finalize)> inheritedWriterGuard(
&inheritedWriterQos, dds_DataWriterQos_finalize);
dds_DataReaderQos inheritedReaderQos{};
if(dds_DataReaderQos_copy(&inheritedReaderQos, &dds_DATAREADER_QOS_DEFAULT) != dds_RETCODE_OK) {
std::cerr << "[DDS ERROR] Could not initialize inherited DataReader QoS.\n";
return;
}
const std::unique_ptr<dds_DataReaderQos, decltype(&dds_DataReaderQos_finalize)> inheritedReaderGuard(
&inheritedReaderQos, dds_DataReaderQos_finalize);
const auto inheritedWriterResult =
dds_xml_QosProfile_get_datawriter_qos(pFactory, pXmlPath, pInheritedProfileName, &inheritedWriterQos);
if(inheritedWriterResult != dds_RETCODE_OK) {
std::cerr << "[DDS ERROR] Could not load DataWriter QoS (code " << inheritedWriterResult << ") for "
<< pInheritedProfileName << " from " << pXmlPath << '\n';
return;
}
const auto inheritedReaderResult =
dds_xml_QosProfile_get_datareader_qos(pFactory, pXmlPath, pInheritedProfileName, &inheritedReaderQos);
if(inheritedReaderResult != dds_RETCODE_OK) {
std::cerr << "[DDS ERROR] Could not load DataReader QoS (code " << inheritedReaderResult << ") for "
<< pInheritedProfileName << " from " << pXmlPath << '\n';
return;
}
const dds::pub::qos::DataWriterQos inheritedProfileWriterQos(inheritedWriterQos);
const dds::sub::qos::DataReaderQos inheritedProfileReaderQos(inheritedReaderQos);
std::cout << "\nBase profile -> Inherited profile\n" << pProfileName << " -> " << pInheritedProfileName << '\n';
helloworld_qosprofile::PrintWriterQos(std::cout, profileWriterQos, inheritedProfileWriterQos);
helloworld_qosprofile::PrintReaderQos(std::cout, profileReaderQos, inheritedProfileReaderQos);
}
} // namespace
int main(int argc, char* argv[]) {
const bool showHelp = argc == 2 && std::string_view(argv[1]) == "--help";
if(showHelp || argc > 4) {
auto& output = showHelp ? std::cout : std::cerr;
output << "Usage: helloworld_qosprofile [xml-path [library::profile [library::inherited-profile]]]\n"
<< "Default XML: " << DEFAULT_XML_PATH << "\nDefault profile: " << DEFAULT_PROFILE
<< "\nInherited profile (no arguments): " << DEFAULT_INHERITED_PROFILE << '\n';
return showHelp ? 0 : -1;
}
const char* pXmlPath = argc >= 2 ? argv[1] : DEFAULT_XML_PATH;
const char* pProfileName = argc >= 3 ? argv[2] : DEFAULT_PROFILE;
const char* pInheritedProfileName = argc == 4 ? argv[3] : nullptr;
if(argc == 1) {
pInheritedProfileName = DEFAULT_INHERITED_PROFILE;
}
try {
Run(pXmlPath, pProfileName, pInheritedProfileName);
return 0;
} catch(const dds::core::Exception& error) {
std::cerr << "[DDS ERROR] " << error.what() << '\n';
return -1;
}
}

void PrintCommonQos(std::ostream& output, const TQos& defaultQos, const TQos& profileQos) {
const auto& defaults = defaultQos.delegate();
const auto& profile = profileQos.delegate();
// The native durability kind also preserves TRANSIENT and PERSISTENT from XML.
PrintValue(output, "durability.kind", FormatValue(defaults.durability.kind),
