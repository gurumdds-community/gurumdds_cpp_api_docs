#include "pub.hpp"
#include "sub.hpp"
#include <gurumdds/cpp/core/Exception.hpp>
#include <gurumdds/cpp/domain/ddsdomain.hpp>
#include <exception>
#include <iostream>
#include <string>
namespace {
inline constexpr dds::domain::DomainId DOMAIN_ID = 0;
inline constexpr char TOPIC_NAME[] = "HelloWorld";
} // namespace
int main(int argc, char* argv[]) {
try {
if(argc == 2) {
const std::string mode = argv[1];
if(mode == "pub") {
helloworld_liveliness::Publisher pub(DOMAIN_ID, TOPIC_NAME);
pub.Run();
return 0;
} else if(mode == "sub") {
helloworld_liveliness::Subscriber sub(DOMAIN_ID, TOPIC_NAME);
sub.Run();
return 0;
}
}
} catch(const dds::core::Exception& error) {
std::cerr << "DDS error: " << error.what() << '\n';
return -1;
} catch(const std::exception& error) {
std::cerr << "Error: " << error.what() << '\n';
return -1;
}
std::cerr << "Usage: helloworld_liveliness <pub|sub>\n";
return -1;
}

#include "pub.hpp"

namespace {
std::mutex g_outputMutex;

namespace helloworld_liveliness {
Publisher::Publisher(dds::domain::DomainId domainId, const std::string& topicName)
: m_participant(domainId),
m_topic(m_participant, topicName),
m_publisher(m_participant, m_participant.default_publisher_qos()
<< dds::core::policy::EntityFactory::ManuallyEnable()),
m_writer(m_publisher, m_topic,
m_publisher.default_writer_qos() << dds::core::policy::Liveliness(
dds::core::policy::LivelinessKind::MANUAL_BY_TOPIC, dds::core::Duration(3, 0))) {}
void Publisher::Run() {
using dds::core::status::StatusMask;
auto mask = StatusMask::publication_matched();
mask |= StatusMask::liveliness_lost();
m_writer.listener(&m_listener, mask);
std::cout << FormatTimestamp(std::chrono::duration_cast<std::chrono::nanoseconds>(
std::chrono::system_clock::now().time_since_epoch())
.count())
<< " [PUB] Liveliness: MANUAL_BY_TOPIC | Lease: 3 s | Press Enter to assert liveliness."
<< " Type q or quit and press Enter to exit." << std::endl;
m_writer.enable();
std::string input;
while(std::getline(std::cin, input)) {
if(input == "q" || input == "quit") {
break;
}
m_writer.assert_liveliness();
const std::lock_guard lock(g_outputMutex);
std::cout << FormatTimestamp(std::chrono::duration_cast<std::chrono::nanoseconds>(
std::chrono::system_clock::now().time_since_epoch())
.count())
<< " [PUB] Asserted liveliness." << std::endl;
}
}
void Publisher::Listener::on_publication_matched(dds::pub::DataWriter<HelloWorld>&,

#ifndef HELLOWORLD_LIVELINESS_TIMESTAMP_HPP
#define HELLOWORLD_LIVELINESS_TIMESTAMP_HPP
#include <chrono>
#include <cstdint>
#include <ctime>
#include <iomanip>
#include <sstream>
#include <string>
namespace helloworld_liveliness {
inline std::string FormatTimestamp(std::int64_t timestamp) {
const auto elapsed = std::chrono::floor<std::chrono::milliseconds>(std::chrono::nanoseconds(timestamp));
const auto seconds = std::chrono::floor<std::chrono::seconds>(elapsed);
const auto milliseconds = elapsed - seconds;
const auto time = static_cast<std::time_t>(seconds.count());
std::tm localTime{};
#ifdef _WIN32
if(localtime_s(&localTime, &time) != 0) {
#else
if(localtime_r(&time, &localTime) == nullptr) {
#endif
return "unavailable";
}
std::ostringstream output;
output << std::put_time(&localTime, "%Y-%m-%d %H:%M:%S") << '.' << std::setfill('0') << std::setw(3)
<< milliseconds.count();
return output.str();
}
} // namespace helloworld_liveliness
#endif // HELLOWORLD_LIVELINESS_TIMESTAMP_HPP

cmake_minimum_required(VERSION 3.20)

find_package(Threads REQUIRED)
set(HELLOWORLD_LIVELINESS_IDL "${CMAKE_CURRENT_SOURCE_DIR}/HelloWorld.idl")
set(HELLOWORLD_LIVELINESS_GENERATED_DIR "${CMAKE_CURRENT_SOURCE_DIR}/generated")
set(HELLOWORLD_LIVELINESS_TYPES_SRCS "${HELLOWORLD_LIVELINESS_GENERATED_DIR}/HelloWorldTypeSupport.cpp")
set(HELLOWORLD_LIVELINESS_TYPES_HEADERS
"${HELLOWORLD_LIVELINESS_GENERATED_DIR}/HelloWorld.hpp"
"${HELLOWORLD_LIVELINESS_GENERATED_DIR}/HelloWorldTypeSupport.hpp"
)
add_custom_command(
OUTPUT "${HELLOWORLD_LIVELINESS_TYPES_SRCS}" ${HELLOWORLD_LIVELINESS_TYPES_HEADERS}
COMMAND "${CMAKE_COMMAND}" -E make_directory "${HELLOWORLD_LIVELINESS_GENERATED_DIR}"
COMMAND "${GURUMIDL}" cpp "${HELLOWORLD_LIVELINESS_IDL}" "${HELLOWORLD_LIVELINESS_GENERATED_DIR}" --replace
DEPENDS "${GURUMIDL}" "${HELLOWORLD_LIVELINESS_IDL}"
COMMENT "Generating HelloWorld C++ types from IDL"
VERBATIM
)
add_custom_target(helloworld_liveliness_generated
DEPENDS "${HELLOWORLD_LIVELINESS_TYPES_SRCS}" ${HELLOWORLD_LIVELINESS_TYPES_HEADERS}
)
add_executable(helloworld_liveliness
${CMAKE_CURRENT_SOURCE_DIR}/src/main.cpp
${CMAKE_CURRENT_SOURCE_DIR}/src/pub.cpp
${CMAKE_CURRENT_SOURCE_DIR}/src/sub.cpp
"${HELLOWORLD_LIVELINESS_TYPES_SRCS}")
add_dependencies(helloworld_liveliness helloworld_liveliness_generated)
target_include_directories(helloworld_liveliness PRIVATE "${HELLOWORLD_LIVELINESS_GENERATED_DIR}")
target_link_libraries(helloworld_liveliness PRIVATE gurumdds::cxx Threads::Threads)
set_target_properties(helloworld_liveliness PROPERTIES OUTPUT_NAME "helloworld_liveliness")
