#include "pub.hpp"
#include "sub.hpp"
#include <gurumdds/cpp/core/Exception.hpp>
#include <gurumdds/cpp/domain/ddsdomain.hpp>
#include <iostream>
#include <string>
namespace {
inline constexpr dds::domain::DomainId DOMAIN_ID = 0;
inline constexpr char TOPIC_NAME[] = "HelloWorld";
} // namespace
int main(int argc, char* argv[]) {
try {
if(argc == 2) {
const std::string mode = argv[1];
if(mode == "pub") {
helloworld_lifespan::Publisher pub(DOMAIN_ID, TOPIC_NAME);
pub.Run();
return 0;
} else if(mode == "sub") {
helloworld_lifespan::Subscriber sub(DOMAIN_ID, TOPIC_NAME);
sub.Run();
return 0;
}
}
} catch(const dds::core::Exception& error) {
std::cerr << "DDS error: " << error.what() << '\n';
return -1;
}
std::cerr << "Usage: helloworld_lifespan <pub|sub>\n";
return -1;
}

src/sub.cpp

#include "sub.hpp"
#include "lifespan.hpp"
#include "timestamp.hpp"
#include <chrono>
#include <iostream>
#include <string>
namespace helloworld_lifespan {
Subscriber::Subscriber(dds::domain::DomainId domainId, const std::string& topicName)
: m_participant(domainId),
m_topic(m_participant, topicName),
m_subscriber(m_participant),
m_reader(m_subscriber, m_topic,
m_subscriber.default_datareader_qos() << dds::core::policy::History::KeepAll()) {}
void Subscriber::Run() {
std::cout << FormatTimestamp(std::chrono::duration_cast<std::chrono::nanoseconds>(
std::chrono::system_clock::now().time_since_epoch())
.count())
<< " [SUB] Lifespan: " << LIFESPAN_SECONDS << " s | Press Enter to take HelloWorld." << std::endl;
std::string input;
while(std::getline(std::cin, input)) {
const auto samples = m_reader.take();
const auto takeTime = m_participant.current_time();
const auto takeTimestamp =
std::chrono::seconds(takeTime.sec()) + std::chrono::nanoseconds(takeTime.nanosec());
std::cout << FormatTimestamp(std::chrono::duration_cast<std::chrono::nanoseconds>(
std::chrono::system_clock::now().time_since_epoch())
.count())
<< " [SUB] Took " << samples.length() << " HelloWorld" << std::endl;
for(const auto& sample: samples) {
const auto& info = sample.info();
const auto sourceTime = info.timestamp();
const auto timestamp =
std::chrono::seconds(sourceTime.sec()) + std::chrono::nanoseconds(sourceTime.nanosec());
const auto elapsed = std::chrono::duration_cast<std::chrono::milliseconds>(takeTimestamp - timestamp);
const bool valid = info.valid();
std::cout << FormatTimestamp(std::chrono::duration_cast<std::chrono::nanoseconds>(
std::chrono::system_clock::now().time_since_epoch())
.count())
<< ' ' << (valid ? " valid=true " : "* valid=false")
<< " | source_timestamp=" << FormatTimestamp(timestamp.count())
<< " | elapsed=" << elapsed.count() << " ms";
if(valid) {
const auto& data = sample.data();
std::cout << " | user_id=" << data.user_id << " | message=\""
<< (data.message != nullptr ? data.message : "null") << '"';
}
std::cout << std::endl;
}
}
}
} // namespace helloworld_lifespan

#ifndef HELLOWORLD_LIFESPAN_TIMESTAMP_HPP
#define HELLOWORLD_LIFESPAN_TIMESTAMP_HPP
#include <chrono>
#include <cstdint>
#include <ctime>
#include <iomanip>
#include <sstream>
#include <string>
namespace helloworld_lifespan {
inline std::string FormatTimestamp(std::int64_t timestamp) {
const auto elapsed = std::chrono::floor<std::chrono::milliseconds>(std::chrono::nanoseconds(timestamp));
const auto seconds = std::chrono::floor<std::chrono::seconds>(elapsed);
const auto milliseconds = elapsed - seconds;
const auto time = static_cast<std::time_t>(seconds.count());
std::tm localTime{};
#ifdef _WIN32
if(localtime_s(&localTime, &time) != 0) {
#else
if(localtime_r(&time, &localTime) == nullptr) {
#endif
return "unavailable";
}
std::ostringstream output;
output << std::put_time(&localTime, "%Y-%m-%d %H:%M:%S") << '.' << std::setfill('0') << std::setw(3)
<< milliseconds.count();
return output.str();
}
} // namespace helloworld_lifespan
#endif // HELLOWORLD_LIFESPAN_TIMESTAMP_HPP

CMakeLists.txt

cmake_minimum_required(VERSION 3.20)
project(helloworld_lifespan VERSION 1.0.0 LANGUAGES CXX)
set(HELLOWORLD_LIFESPAN_IDL "${CMAKE_CURRENT_SOURCE_DIR}/HelloWorld.idl")
set(HELLOWORLD_LIFESPAN_GENERATED_DIR "${CMAKE_CURRENT_SOURCE_DIR}/generated")
set(HELLOWORLD_LIFESPAN_TYPES_SRCS "${HELLOWORLD_LIFESPAN_GENERATED_DIR}/HelloWorldTypeSupport.cpp")
set(HELLOWORLD_LIFESPAN_TYPES_HEADERS
"${HELLOWORLD_LIFESPAN_GENERATED_DIR}/HelloWorld.hpp"
"${HELLOWORLD_LIFESPAN_GENERATED_DIR}/HelloWorldTypeSupport.hpp"
)
add_custom_command(
OUTPUT "${HELLOWORLD_LIFESPAN_TYPES_SRCS}" ${HELLOWORLD_LIFESPAN_TYPES_HEADERS}
COMMAND "${CMAKE_COMMAND}" -E make_directory "${HELLOWORLD_LIFESPAN_GENERATED_DIR}"
COMMAND "${GURUMIDL}" cpp "${HELLOWORLD_LIFESPAN_IDL}" "${HELLOWORLD_LIFESPAN_GENERATED_DIR}" --replace
DEPENDS "${GURUMIDL}" "${HELLOWORLD_LIFESPAN_IDL}"
COMMENT "Generating HelloWorld C++ types from IDL"
VERBATIM
)
add_custom_target(helloworld_lifespan_generated
DEPENDS "${HELLOWORLD_LIFESPAN_TYPES_SRCS}" ${HELLOWORLD_LIFESPAN_TYPES_HEADERS}
)
add_executable(helloworld_lifespan
${CMAKE_CURRENT_SOURCE_DIR}/src/main.cpp
${CMAKE_CURRENT_SOURCE_DIR}/src/pub.cpp
${CMAKE_CURRENT_SOURCE_DIR}/src/sub.cpp
"${HELLOWORLD_LIFESPAN_TYPES_SRCS}")
add_dependencies(helloworld_lifespan helloworld_lifespan_generated)
target_include_directories(helloworld_lifespan PRIVATE "${HELLOWORLD_LIFESPAN_GENERATED_DIR}")
target_link_libraries(helloworld_lifespan PRIVATE gurumdds::cxx)
set_target_properties(helloworld_lifespan PROPERTIES OUTPUT_NAME "helloworld_lifespan")
