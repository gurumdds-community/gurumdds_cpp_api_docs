Keyed sample

여러 instance의 sample을 발행하고 user ID로 특정 instance를 선택합니다.

예제 실행하기

./helloworld_keyed sub 1
./helloworld_keyed pub 2

소스 파일

HelloWorld.idl

struct HelloWorld {
@key long user_id;
string message;
};

src/main.cpp

#include "pub.hpp"
#include "sub.hpp"
#include <gurumdds/cpp/core/Exception.hpp>
#include <gurumdds/cpp/domain/ddsdomain.hpp>
#include <charconv>
#include <iostream>
#include <string_view>
namespace {
inline constexpr dds::domain::DomainId DOMAIN_ID = 0;
inline constexpr char TOPIC_NAME[] = "HelloWorld";
} // namespace
int main(int argc, char* argv[]) {
if(argc != 3) {
std::cerr << "Usage: helloworld_keyed pub <instance_count>\n"
<< " helloworld_keyed sub <user_id> (user_id >= 1)\n";
return -1;
}
const std::string_view mode = argv[1];
if(mode != "pub" && mode != "sub") {
std::cerr << "[ERROR] Unknown mode: " << mode << " (expected pub or sub)\n";
return -1;
}
const std::string_view input = argv[2];
dds_Long value = 0;
const char* pEnd = input.data() + input.size();
const auto result = std::from_chars(input.data(), pEnd, value);
if(result.ec != std::errc{} || result.ptr != pEnd || value < 1) {
std::cerr << (mode == "pub" ? "[PUB ERROR] instance_count must be a positive 32-bit integer.\n"
: "[SUB ERROR] user_id must be a positive 32-bit integer.\n");
return -1;
}
try {
if(mode == "pub") {
helloworld::Publisher pub(DOMAIN_ID, TOPIC_NAME);
pub.Run(value);
return 0;
}
helloworld::Subscriber sub(DOMAIN_ID, TOPIC_NAME);
sub.Run(value);
return 0;
} catch(const dds::core::Exception& error) {
std::cerr << "[DDS ERROR] " << error.what() << '\n';
return -1;
}
}

#include "pub.hpp"
#include <chrono>
#include <iostream>
#include <thread>
#include <vector>
namespace helloworld {
Publisher::Publisher(dds::domain::DomainId domainId, const std::string& topicName)
: m_participant(domainId),
m_topic(m_participant, topicName),
m_publisher(m_participant),
m_writer(m_publisher, m_topic) {}
void Publisher::Run(dds_Long instanceCount) {
while(m_writer.publication_matched_status().current_count() == 0) {
std::this_thread::sleep_for(std::chrono::milliseconds(100));
}
std::cout << "=== PUB | Register instances ===" << std::endl;
for(dds_Long i = 0; i < instanceCount; ++i) {
const HelloWorld sample{i + 1, nullptr};
const dds::core::InstanceHandle instanceHandle = m_writer.register_instance(sample);
if(instanceHandle.is_nil()) {
std::cerr << "[PUB ERROR] Could not register HelloWorld | user_id=" << sample.user_id << std::endl;
return;
}
std::cout << " [" << i + 1 << "] user_id=" << sample.user_id << " | instance_handle=" << instanceHandle
<< std::endl;
}
std::cout << "\n=== PUB | Look up instances ===" << std::endl;
std::vector<dds::core::InstanceHandle> instances;
for(dds_Long i = 0; i < instanceCount; ++i) {
const HelloWorld sample{i + 1, nullptr};
const dds::core::InstanceHandle instanceHandle = m_writer.lookup_instance(sample);
if(instanceHandle.is_nil()) {
std::cerr << "[PUB ERROR] Could not look up HelloWorld | user_id=" << sample.user_id << std::endl;
return;
}
std::cout << " [" << i + 1 << "] user_id=" << sample.user_id << " | instance_handle=" << instanceHandle
<< std::endl;
instances.push_back(instanceHandle);
}
std::cout << "\n=== PUB | Read instance keys ===" << std::endl;
for(dds_Long i = 0; i < instanceCount; ++i) {
HelloWorld sample{};
m_writer.key_value(sample, instances[i]);
std::cout << " [" << i + 1 << "] user_id=" << sample.user_id << " | instance_handle=" << instances[i]
<< std::endl;
}
std::cout << "\n=== PUB | Write samples ===" << std::endl;
std::this_thread::sleep_for(std::chrono::milliseconds(1000));
char message[] = "Hello!";
for(dds_Long i = 0; i < instanceCount; ++i) {
const auto& instanceHandle = instances[i];
/* When instanceHandle is set to nil during a write operation, the instance is automatically
* registered internally, or an existing registered instance is retrieved.
* e.g. m_writer.write(sample, dds::core::InstanceHandle::nil());
*/
const HelloWorld sample{i + 1, message};
m_writer.write(sample, instanceHandle);
std::cout << " [" << i + 1 << "] user_id=" << sample.user_id << " | instance_handle=" << instanceHandle
<< " | message=\"" << sample.message << '"' << std::endl;
}
std::this_thread::sleep_for(std::chrono::milliseconds(1000));
std::cout << "\n=== PUB | Dispose instances ===" << std::endl;
for(dds_Long i = 0; i < instanceCount; ++i) {
HelloWorld sample{};
m_writer.key_value(sample, instances[i]);
m_writer.dispose_instance(instances[i]);
std::cout << " [" << i + 1 << "] user_id=" << sample.user_id << " | instance_handle=" << instances[i]
<< std::endl;
}
std::this_thread::sleep_for(std::chrono::milliseconds(1000));
std::cout << "\n=== PUB | Unregister instances ===" << std::endl;
for(dds_Long i = 0; i < instanceCount; ++i) {
HelloWorld sample{};
m_writer.key_value(sample, instances[i]);
m_writer.unregister_instance(instances[i]);
std::cout << " [" << i + 1 << "] user_id=" << sample.user_id << " | instance_handle=" << instances[i]
<< std::endl;
}
/* Wait until the subscriber receives the data and terminates.
* If the publisher application exits immediately without waiting, it
* may terminate before the data is actually transmitted.
*/
while(m_writer.publication_matched_status().current_count() > 0) {
std::this_thread::sleep_for(std::chrono::milliseconds(100));
}
}
} // namespace helloworld

#include "sub.hpp"
#include <chrono>
#include <iostream>
#include <thread>
namespace helloworld {
Subscriber::Subscriber(dds::domain::DomainId domainId, const std::string& topicName)
: m_participant(domainId),
m_topic(m_participant, topicName),
m_subscriber(m_participant),
m_reader(m_subscriber, m_topic,
/* Without this QoS setting, the instance may be deleted immediately,
* preventing the sample from being taken.
*/
m_subscriber.default_datareader_qos()
<< dds::core::policy::ReaderDataLifecycle::NoAutoPurgeDisposedSamples()) {}
void Subscriber::Run(dds_Long userId) {
const HelloWorld key{userId, nullptr};
std::cout << "[SUB] Waiting for user_id=" << userId << std::endl;
auto instanceHandle = m_reader.lookup_instance(key);
while(instanceHandle.is_nil()) {
instanceHandle = m_reader.lookup_instance(key);
std::this_thread::sleep_for(std::chrono::milliseconds(100));
}
bool isRunning = true;
while(isRunning) {
for(const auto& sample: m_reader.select().instance(instanceHandle).take()) {
const auto& info = sample.info();
std::cout << "[SUB] HelloWorld | key(user_id)=" << userId
<< " | instance_handle=" << info.instance_handle() << std::endl;
if(info.valid()) {
const auto& data = sample.data();
std::cout << " Data | user_id=" << data.user_id << " | message=\"" << data.message << '"'
<< std::endl;
}
const auto instanceState = info.state().instance_state();
if(instanceState == dds::sub::status::InstanceState::alive()) {
std::cout << " State | ALIVE" << std::endl;
} else if(instanceState == dds::sub::status::InstanceState::not_alive_disposed()) {
std::cout << " State | NOT_ALIVE_DISPOSED" << std::endl;
} else if(instanceState == dds::sub::status::InstanceState::not_alive_no_writers()) {
std::cout << " State | NOT_ALIVE_NO_WRITERS" << std::endl;
isRunning = false;
}
std::cout << std::endl;
}
std::this_thread::sleep_for(std::chrono::milliseconds(100));
}
}
} // namespace helloworld

CMakeLists.txt

cmake_minimum_required(VERSION 3.20)
project(helloworld_keyed VERSION 1.0.0 LANGUAGES CXX)
set(HELLOWORLD_KEYED_IDL "${CMAKE_CURRENT_SOURCE_DIR}/HelloWorld.idl")
set(HELLOWORLD_KEYED_GENERATED_DIR "${CMAKE_CURRENT_SOURCE_DIR}/generated")
set(HELLOWORLD_KEYED_TYPES_SRCS "${HELLOWORLD_KEYED_GENERATED_DIR}/HelloWorldTypeSupport.cpp")
set(HELLOWORLD_KEYED_TYPES_HEADERS
"${HELLOWORLD_KEYED_GENERATED_DIR}/HelloWorld.hpp"
"${HELLOWORLD_KEYED_GENERATED_DIR}/HelloWorldTypeSupport.hpp"
)
add_custom_command(
OUTPUT "${HELLOWORLD_KEYED_TYPES_SRCS}" ${HELLOWORLD_KEYED_TYPES_HEADERS}
COMMAND "${CMAKE_COMMAND}" -E make_directory "${HELLOWORLD_KEYED_GENERATED_DIR}"
COMMAND "${GURUMIDL}" cpp "${HELLOWORLD_KEYED_IDL}" "${HELLOWORLD_KEYED_GENERATED_DIR}" --replace
DEPENDS "${GURUMIDL}" "${HELLOWORLD_KEYED_IDL}"
COMMENT "Generating HelloWorld C++ types from IDL"
VERBATIM
)
add_custom_target(helloworld_keyed_generated
DEPENDS "${HELLOWORLD_KEYED_TYPES_SRCS}" ${HELLOWORLD_KEYED_TYPES_HEADERS}
)
add_executable(helloworld_keyed
${CMAKE_CURRENT_SOURCE_DIR}/src/main.cpp
${CMAKE_CURRENT_SOURCE_DIR}/src/pub.cpp
${CMAKE_CURRENT_SOURCE_DIR}/src/sub.cpp
"${HELLOWORLD_KEYED_TYPES_SRCS}")
add_dependencies(helloworld_keyed helloworld_keyed_generated)
target_include_directories(helloworld_keyed PRIVATE "${HELLOWORLD_KEYED_GENERATED_DIR}")
target_link_libraries(helloworld_keyed PRIVATE gurumdds::cxx)
set_target_properties(helloworld_keyed PROPERTIES OUTPUT_NAME "helloworld_keyed")
