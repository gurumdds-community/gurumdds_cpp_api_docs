DataReader listener

DataReader listener로 data-available 알림을 받습니다.

예제 실행하기

./helloworld_listener sub
./helloworld_listener pub

HelloWorld.idl

struct HelloWorld {
long user_id;
string message;
};

#include "pub.hpp"
#include "sub.hpp"
#include <gurumdds/cpp/core/Exception.hpp>

#include <exception>
#include <iostream>
#include <string>
namespace {
inline constexpr dds::domain::DomainId DOMAIN_ID = 0;
inline constexpr char TOPIC_NAME[] = "HelloWorld";
} // namespace
int main(int argc, char* argv[]) {
try {
if(argc == 2) {
const std::string mode = argv[1];
if(mode == "pub") {
helloworld_listener::Publisher pub(DOMAIN_ID, TOPIC_NAME);
pub.Run();
return 0;
} else if(mode == "sub") {
helloworld_listener::Subscriber sub(DOMAIN_ID, TOPIC_NAME);
sub.Run();
return 0;
}
}
} catch(const dds::core::Exception& error) {
std::cerr << "DDS error: " << error.what() << '\n';
return -1;
} catch(const std::exception& error) {
std::cerr << "Error: " << error.what() << '\n';
return -1;
}
std::cerr << "Usage: helloworld_listener <pub|sub>\n";
return -1;
}

#include "pub.hpp"
#include <chrono>
#include <iostream>
#include <thread>
namespace helloworld_listener {
Publisher::Publisher(dds::domain::DomainId domainId, const std::string& topicName)
: m_participant(domainId),
m_topic(m_participant, topicName),
m_publisher(m_participant),
m_writer(m_publisher, m_topic) {}
void Publisher::Run() {
while(m_writer.publication_matched_status().current_count() == 0) {
std::this_thread::sleep_for(std::chrono::milliseconds(100));
}
char message[] = "Hello!";
const HelloWorld sample{0, message};
m_writer.write(sample);
std::cout << "Sent HelloWorld" << std::endl;
std::cout << "user_id: " << sample.user_id << std::endl;
std::cout << "message: " << sample.message << std::endl;
/* Wait until the subscriber receives the data and terminates.
* If the publisher application exits immediately without waiting, it
* may terminate before the data is actually transmitted.
*/
while(m_writer.publication_matched_status().current_count() > 0) {
std::this_thread::sleep_for(std::chrono::milliseconds(100));
}
}
} // namespace helloworld_listener

src/sub.cpp

#include "sub.hpp"
#include <exception>
#include <iostream>
namespace helloworld_listener {
Subscriber::Subscriber(dds::domain::DomainId domainId, const std::string& topicName)
: m_participant(domainId),
m_topic(m_participant, topicName),
m_subscriber(m_participant, m_participant.default_subscriber_qos()
<< dds::core::policy::EntityFactory::ManuallyEnable()),
m_reader(m_subscriber, m_topic) {}
void Subscriber::Run() {
// Install the listener before the reader can match a publisher and receive samples.
m_reader.listener(&m_listener, dds::core::status::StatusMask::data_available());
m_reader.enable();
m_listener.Wait();
}
void Subscriber::Listener::Wait() {
m_completion.get_future().get();
}
void Subscriber::Listener::on_data_available(dds::sub::DataReader<T>& reader) {
try {
bool received = false;
for(const auto& sample: reader.take()) {
if(!sample.info().valid()) {
continue;
}
const auto& data = sample.data();
std::cout << "Received HelloWorld" << std::endl;
std::cout << "user_id: " << data.user_id << std::endl;
std::cout << "message: " << (data.message != nullptr ? data.message : "null") << std::endl;
received = true;
break;
}
// The loan from take() has been returned before waking Run().
if(received && !m_completed.exchange(true)) {
m_completion.set_value();
}
} catch(...) {
if(!m_completed.exchange(true)) {
m_completion.set_exception(std::current_exception());
}
}
}
} // namespace helloworld_listener

#define HELLOWORLD_LISTENER_SUB_HPP

#include <string>

Subscriber(dds::domain::DomainId domainId, const std::string& topicName);

void on_data_available(dds::sub::DataReader<T>& reader) override;

dds::topic::Topic<T> m_topic;

} // namespace helloworld_listener

CMakeLists.txt

cmake_minimum_required(VERSION 3.20)
project(helloworld_listener VERSION 1.0.0 LANGUAGES CXX)
find_package(Threads REQUIRED)
set(HELLOWORLD_LISTENER_IDL "${CMAKE_CURRENT_SOURCE_DIR}/HelloWorld.idl")
set(HELLOWORLD_LISTENER_GENERATED_DIR "${CMAKE_CURRENT_SOURCE_DIR}/generated")
set(HELLOWORLD_LISTENER_TYPES_SRCS "${HELLOWORLD_LISTENER_GENERATED_DIR}/HelloWorldTypeSupport.cpp")
set(HELLOWORLD_LISTENER_TYPES_HEADERS
"${HELLOWORLD_LISTENER_GENERATED_DIR}/HelloWorld.hpp"
"${HELLOWORLD_LISTENER_GENERATED_DIR}/HelloWorldTypeSupport.hpp"
)
add_custom_command(
OUTPUT "${HELLOWORLD_LISTENER_TYPES_SRCS}" ${HELLOWORLD_LISTENER_TYPES_HEADERS}
COMMAND "${CMAKE_COMMAND}" -E make_directory "${HELLOWORLD_LISTENER_GENERATED_DIR}"
COMMAND "${GURUMIDL}" cpp "${HELLOWORLD_LISTENER_IDL}" "${HELLOWORLD_LISTENER_GENERATED_DIR}" --replace
DEPENDS "${GURUMIDL}" "${HELLOWORLD_LISTENER_IDL}"
COMMENT "Generating HelloWorld C++ types from IDL"
VERBATIM
)
add_custom_target(helloworld_listener_generated
DEPENDS "${HELLOWORLD_LISTENER_TYPES_SRCS}" ${HELLOWORLD_LISTENER_TYPES_HEADERS}
)
add_executable(helloworld_listener
${CMAKE_CURRENT_SOURCE_DIR}/src/main.cpp
${CMAKE_CURRENT_SOURCE_DIR}/src/pub.cpp
${CMAKE_CURRENT_SOURCE_DIR}/src/sub.cpp
"${HELLOWORLD_LISTENER_TYPES_SRCS}")
add_dependencies(helloworld_listener helloworld_listener_generated)
target_include_directories(helloworld_listener PRIVATE "${HELLOWORLD_LISTENER_GENERATED_DIR}")
target_link_libraries(helloworld_listener PRIVATE gurumdds::cxx Threads::Threads)
set_target_properties(helloworld_listener PROPERTIES OUTPUT_NAME "helloworld_listener")
