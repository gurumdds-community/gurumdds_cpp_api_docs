Reads unread samples, looks up their handles, inspects instance state changes, and takes samples for one instance

See Receive and select samples for the expected instance state sequence.

HelloWorld.idl

module Hello {
    struct Msg {
        @key
        long userId;
        string message;
    };
};

Keyed Publisher

#include <chrono>
#include <thread>
#include <gurumdds/dds.hpp>
#include <HelloWorldTypeSupport.hpp>

int main(int argc, char** argv) {
    try {
        dds::domain::DomainId domain_id = 79;
        dds::domain::DomainParticipant dp(domain_id);
        dds::topic::Topic<Hello::Msg> topic(dp, "HelloWorld");
        dds::pub::Publisher pub(dp);

        dds::pub::qos::DataWriterQos writer_qos;
        writer_qos->writer_data_lifecycle.autodispose_unregistered_instances = false;

        dds::pub::DataWriter<Hello::Msg> dw(pub, topic, writer_qos);

        Hello::Msg a_sample;
        Hello::Msg b_sample;
        Hello::Msg c_sample;

        a_sample.userId = 1;
        a_sample.message = const_cast<char*>("hello");

        b_sample.userId = 2;
        b_sample.message = const_cast<char*>("foo");

        c_sample.userId = 3;
        c_sample.message = const_cast<char*>("bar");

        dds::core::InstanceHandle a_handle = dw.register_instance(a_sample);
        if (a_handle == dds::core::InstanceHandle(dds::core::InstanceHandle::nil()))
            throw std::exception();

        if (a_handle != dw.lookup_instance(a_sample))
            throw std::exception();

        Hello::Msg keyholder;
        dw.key_value(keyholder, a_handle);
        if (keyholder.userId != a_sample.userId)
            throw std::exception();

        dds_Time_t delay = {1, 500 * 1000 * 1000};
        dds_Time_sleep(&delay);

        dw.write(a_sample, a_handle);
        dw.write(b_sample);
        dw.write(c_sample, dds::core::InstanceHandle::nil());

        dds_Time_sleep(&delay);

        dw.dispose_instance(a_handle);

        dds_Time_sleep(&delay);

        a_sample.message = const_cast<char*>("world");
        dw.write(a_sample);

        dds_Time_sleep(&delay);

        dw.unregister_instance(a_handle);

        delay.sec = 3;
        dds_Time_sleep(&delay);
    } catch(const dds::core::Error& e) {
        std::cout << e.what();
    }

    dds_DomainParticipantFactory_shutdown();

    return 0;
}

Keyed Subscriber

#include <chrono>
#include <thread>
#include <gurumdds/dds.hpp>
#include <HelloWorldTypeSupport.hpp>

int main(int argc, char** argv) {
    try {
        dds::domain::DomainId domain_id = 79;
        dds::domain::DomainParticipant dp(domain_id);
        dds::topic::Topic<Hello::Msg> topic(dp, "HelloWorld");
        dds::sub::Subscriber sub(dp);

        dds::sub::qos::DataReaderQos reader_qos;
        reader_qos->reader_data_lifecycle = {
            .autopurge_nowriter_samples_delay = {
                .sec = dds_DURATION_INFINITE_SEC,
                .nanosec = dds_DURATION_INFINITE_NSEC,
            },
            .autopurge_disposed_samples_delay = {
                .sec = dds_DURATION_INFINITE_SEC,
                .nanosec = dds_DURATION_INFINITE_NSEC},
        };

        dds::sub::DataReader<Hello::Msg> dr(sub, topic, reader_qos);
        dds::sub::AnyDataReader reader{dr};
        auto dr2 = reader.get<Hello::Msg>();

        dds::core::InstanceHandle last_lookup_handle(dds::core::InstanceHandle::nil());
        bool is_running = true;
        while (is_running) {
            auto samples = dr.select()
                .max_samples(8)
                .state(dds::sub::status::DataState::any() << dds::sub::status::SampleState::not_read())
                .read();

            dds::sub::status::DataState::any();

            if (samples.length() == 0) {
                dds_Time_t delay = {1, 0};
                dds_Time_sleep(&delay);
                continue;
            }

            for (const auto& sample : samples) {
                std::cout << "Data received!" << std::endl;
                if (sample.info().valid()) {
                    last_lookup_handle = dr.lookup_instance(sample.data());
                    if (last_lookup_handle != sample.info().instance_handle())
                        throw std::exception();

                    std::cout << "userId: " << sample->data().userId << std::endl;
                    std::cout << "message: " << sample->data().message << std::endl;
                }

                Hello::Msg keyholder;
                dr.key_value(keyholder, sample.info().instance_handle());
                switch (sample.info().state().instance_state().to_ulong()) {
                    case dds_ALIVE_INSTANCE_STATE:
                        std::cout << "key: " << keyholder.userId
                                    << ", instance state: ALIVE, instance handle: " << static_cast<uint64_t>(sample.info().instance_handle())
                                    << std::endl;
                        break;
                    case dds_NOT_ALIVE_DISPOSED_INSTANCE_STATE:
                        std::cout << "key: " << keyholder.userId
                                    << ", instance state: NOT_ALIVE_DISPOSED, instance handle: " << static_cast<uint64_t>(sample.info().instance_handle())
                                    << std::endl;
                        break;
                    case dds_NOT_ALIVE_NO_WRITERS_INSTANCE_STATE:
                        std::cout << "key: " << keyholder.userId
                                    << ", instance state: NOT_ALIVE_NO_WRITERS, instance handle: " << static_cast<uint64_t>(sample.info().instance_handle())
                                    << std::endl;
                        is_running = false;
                        break;
                    default:
                        throw std::exception();
                }
                std::cout << std::endl;
            }
        }

        auto samples = dr.select()
            .max_samples(8)
            .instance(last_lookup_handle)
            .take();

        if (samples.length() == 0)
            throw std::exception();

        std::cout << "take_instance(handle = " << static_cast<uint64_t>(last_lookup_handle) << ")" << std::endl;
        for (const auto& sample : samples) {
            if (sample.info().valid()) {
                std::cout << "userId: " << sample->data().userId << std::endl;
                std::cout << "message: " << sample->data().message << std::endl;
            }
        }
    } catch(const dds::core::Error& e) {
        std::cout << "an error occurred: " << e.what() << std::endl;
    }

    dds_Time_t delay = { 1, 500 * 1000 * 1000 };
    dds_Time_sleep(&delay);

    // Terminate the GurumDDS middleware.
    // This API will send a messages that left a communication.
    // And Dispose of any resources you used in the GurumDDS.
    dds_DomainParticipantFactory_shutdown();

    return 0;
}
