src/main.cpp

#include "pub.hpp"
#include "sub.hpp"

#include <gurumdds/cpp/core/Exception.hpp>
#include <gurumdds/cpp/domain/ddsdomain.hpp>

#include <exception>
#include <iostream>
#include <string_view>

namespace {

    inline constexpr dds::domain::DomainId DOMAIN_ID = 0;
    inline constexpr char TOPIC_NAME[] = "HelloWorld";

} // namespace

int main(int argc, char* argv[]) {
    try {
        if(argc == 2) {
            const std::string_view mode = argv[1];
            if(mode == "pub") {
                helloworld_timebasedfilter::Publisher pub(DOMAIN_ID, TOPIC_NAME);
                pub.Run();
                return 0;
            }
            if(mode == "sub") {
                helloworld_timebasedfilter::Subscriber sub(DOMAIN_ID, TOPIC_NAME);
                sub.Run();
                return 0;
            }
        }
    } catch(const dds::core::Exception& error) {
        std::cerr << "DDS error: " << error.what() << '\n';
        return -1;
    } catch(const std::exception& error) {
        std::cerr << "Error: " << error.what() << '\n';
        return -1;
    }

    std::cerr << "Usage: helloworld_timebasedfilter <pub|sub>\n";
    return -1;
}

#include <chrono>

namespace helloworld_timebasedfilter {

    Publisher::Publisher(dds::domain::DomainId domainId, const std::string& topicName)
        : m_participant(domainId),
          m_topic(m_participant, topicName),
          m_publisher(m_participant),
          m_writer(m_publisher, m_topic) {}

    void Publisher::Run() {
        std::cout << FormatTimestamp(std::chrono::duration_cast<std::chrono::nanoseconds>(
                                         std::chrono::system_clock::now().time_since_epoch())
                                         .count())
                  << " [PUB] Press Enter to send HelloWorld." << std::endl;

        char message[] = "Hello!";
        const HelloWorld sample{0, message};
        std::string input;
        while(std::getline(std::cin, input)) {
            m_writer.write(sample);
            std::cout << FormatTimestamp(std::chrono::duration_cast<std::chrono::nanoseconds>(
                                             std::chrono::system_clock::now().time_since_epoch())
                                             .count())
                      << " [PUB] Sent HelloWorld | user_id=" << sample.user_id << " | message=\"" << sample.message
                      << '"' << std::endl;
        }
    }

} // namespace helloworld_timebasedfilter

src/sub.cpp

#include "sub.hpp"
#include "timestamp.hpp"

#include <chrono>
#include <iostream>
#include <thread>

namespace helloworld_timebasedfilter {

    Subscriber::Subscriber(dds::domain::DomainId domainId, const std::string& topicName)
        : m_participant(domainId),
          m_topic(m_participant, topicName),
          m_subscriber(m_participant, m_participant.default_subscriber_qos()
                                          << dds::core::policy::EntityFactory::ManuallyEnable()),
          m_reader(m_subscriber, m_topic,
                   m_subscriber.default_datareader_qos()
                       << dds::core::policy::TimeBasedFilter(dds::core::Duration(1, 0))) {}

    void Subscriber::Run() {
        // Install the listener before the reader can match a publisher and receive samples.
        m_reader.listener(&m_listener, dds::core::status::StatusMask::data_available());
        const auto minimumSeparation = m_reader.qos().policy<dds::core::policy::TimeBasedFilter>().minimum_separation();
        std::cout << FormatTimestamp(std::chrono::duration_cast<std::chrono::nanoseconds>(
                                         std::chrono::system_clock::now().time_since_epoch())
                                         .count())
                  << " [SUB] Minimum separation: " << minimumSeparation.to_millisecs()
                  << " ms | Waiting for HelloWorld." << std::endl;
        m_reader.enable();

        while(true) {
            std::this_thread::sleep_for(std::chrono::seconds(1));
        }
    }

    void Subscriber::Listener::on_data_available(dds::sub::DataReader<HelloWorld>& reader) {
        for(const auto& sample: reader.take()) {
            if(!sample.info().valid()) {
                continue;
            }

            const auto& data = sample.data();
            std::cout << FormatTimestamp(std::chrono::duration_cast<std::chrono::nanoseconds>(
                                             std::chrono::system_clock::now().time_since_epoch())
                                             .count())
                      << " [SUB] Received HelloWorld | user_id=" << data.user_id << " | message=\""
                      << (data.message != nullptr ? data.message : "null") << '"' << std::endl;
        }
    }

} // namespace helloworld_timebasedfilter

#ifndef HELLOWORLD_TIMEBASEDFILTER_SUB_HPP
#define HELLOWORLD_TIMEBASEDFILTER_SUB_HPP

#include "HelloWorldTypeSupport.hpp"
#include "gurumdds/dds.hpp"

#include <string>

namespace helloworld_timebasedfilter {

    class Subscriber {
    public:
        Subscriber(dds::domain::DomainId domainId, const std::string& topicName);
        void Run();

    private:
        class Listener: public dds::sub::NoOpDataReaderListener<HelloWorld> {
            void on_data_available(dds::sub::DataReader<HelloWorld>& reader) override;
        };

        // The listener must outlive the DDS entities that can invoke it.
        Listener m_listener;
        dds::domain::DomainParticipant m_participant;
        dds::topic::Topic<HelloWorld> m_topic;
        dds::sub::Subscriber m_subscriber;
        dds::sub::DataReader<HelloWorld> m_reader;

} // namespace helloworld_timebasedfilter

src/timestamp.hpp

#ifndef HELLOWORLD_TIMEBASEDFILTER_TIMESTAMP_HPP
#define HELLOWORLD_TIMEBASEDFILTER_TIMESTAMP_HPP

#include <chrono>
#include <cstdint>
#include <ctime>
#include <iomanip>
#include <sstream>
#include <string>

namespace helloworld_timebasedfilter {

    inline std::string FormatTimestamp(std::int64_t timestamp) {
        const auto elapsed = std::chrono::floor<std::chrono::milliseconds>(std::chrono::nanoseconds(timestamp));
        const auto seconds = std::chrono::floor<std::chrono::seconds>(elapsed);
        const auto milliseconds = elapsed - seconds;
        const auto time = static_cast<std::time_t>(seconds.count());
        std::tm localTime{};
#ifdef _WIN32
        if(localtime_s(&localTime, &time) != 0) {
#else
        if(localtime_r(&time, &localTime) == nullptr) {
#endif
            return "unavailable";
        }

        std::ostringstream output;
        output << std::put_time(&localTime, "%Y-%m-%d %H:%M:%S") << '.' << std::setfill('0') << std::setw(3)
               << milliseconds.count();
        return output.str();
    }

} // namespace helloworld_timebasedfilter

#endif // HELLOWORLD_TIMEBASEDFILTER_TIMESTAMP_HPP

CMakeLists.txt

cmake_minimum_required(VERSION 3.20)
project(helloworld_timebasedfilter VERSION 1.0.0 LANGUAGES CXX)

find_package(Threads REQUIRED)

set(HELLOWORLD_TIMEBASEDFILTER_IDL "${CMAKE_CURRENT_SOURCE_DIR}/HelloWorld.idl")
set(HELLOWORLD_TIMEBASEDFILTER_GENERATED_DIR "${CMAKE_CURRENT_SOURCE_DIR}/generated")
set(HELLOWORLD_TIMEBASEDFILTER_TYPES_SRCS "${HELLOWORLD_TIMEBASEDFILTER_GENERATED_DIR}/HelloWorldTypeSupport.cpp")
set(HELLOWORLD_TIMEBASEDFILTER_TYPES_HEADERS
    "${HELLOWORLD_TIMEBASEDFILTER_GENERATED_DIR}/HelloWorld.hpp"
    "${HELLOWORLD_TIMEBASEDFILTER_GENERATED_DIR}/HelloWorldTypeSupport.hpp"
)

add_custom_command(
    OUTPUT "${HELLOWORLD_TIMEBASEDFILTER_TYPES_SRCS}" ${HELLOWORLD_TIMEBASEDFILTER_TYPES_HEADERS}
    COMMAND "${CMAKE_COMMAND}" -E make_directory "${HELLOWORLD_TIMEBASEDFILTER_GENERATED_DIR}"
    COMMAND "${GURUMIDL}" cpp "${HELLOWORLD_TIMEBASEDFILTER_IDL}" "${HELLOWORLD_TIMEBASEDFILTER_GENERATED_DIR}" --replace
    DEPENDS "${GURUMIDL}" "${HELLOWORLD_TIMEBASEDFILTER_IDL}"
    COMMENT "Generating HelloWorld C++ types from IDL"
    VERBATIM
)

add_custom_target(helloworld_timebasedfilter_generated
    DEPENDS "${HELLOWORLD_TIMEBASEDFILTER_TYPES_SRCS}" ${HELLOWORLD_TIMEBASEDFILTER_TYPES_HEADERS}
)

add_executable(helloworld_timebasedfilter
    ${CMAKE_CURRENT_SOURCE_DIR}/src/main.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/src/pub.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/src/sub.cpp
    "${HELLOWORLD_TIMEBASEDFILTER_TYPES_SRCS}")
add_dependencies(helloworld_timebasedfilter helloworld_timebasedfilter_generated)
target_include_directories(helloworld_timebasedfilter PRIVATE
    "${HELLOWORLD_TIMEBASEDFILTER_GENERATED_DIR}"
)
target_link_libraries(helloworld_timebasedfilter PRIVATE gurumdds::cxx Threads::Threads)
set_target_properties(helloworld_timebasedfilter PROPERTIES OUTPUT_NAME "helloworld_timebasedfilter")
