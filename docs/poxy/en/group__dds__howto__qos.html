Choose QoS before creating the entities. Start from the Publisher and Subscriber defaults and change the policies needed for this exchange. Check matching before testing delivery, since incompatible endpoint policies can prevent a match.

The Publisher and Subscriber tutorials showed how an endpoint can inherit defaults. Here start with those defaults, set Reliability on both sides, and pass the resulting QoS values when creating the endpoints.

Configure reliable endpoints

Start with the publisher's and subscriber's defaults. Set Reliability and History on both endpoints before constructing them. The writer offers reliable delivery and the reader requests it.

In this example, KeepLast(4) keeps up to four samples per instance, subject to the endpoint's resource limits. The writer's 200 ms blocking limit applies when a reliable write must wait for space.

using dds::core::policy::History;
using dds::core::policy::Reliability;
const auto reliable = Reliability::Reliable(dds::core::Duration::from_millisecs(200));

auto writer_qos = publisher.default_writer_qos();
writer_qos << reliable << History::KeepLast(4);
auto reader_qos = subscriber.default_datareader_qos();
reader_qos << reliable << History::KeepLast(4);

dds::pub::DataWriter<Hello::Msg> writer(publisher, topic, writer_qos);
dds::sub::DataReader<Hello::Msg> reader(subscriber, topic, reader_qos);

Read a single QoS policy

Extract History with operator>> and inspect its depth.

History history;
writer.qos() >> history;
require(history.depth() == 3, "Could not extract writer history policy");

Detect incompatible reliability

A reader requesting reliable delivery cannot match a writer offering best-effort delivery. Check the incompatible QoS counters at both endpoints and use the match count to verify the connection state.

auto writer_qos = publisher.default_writer_qos();
writer_qos << Reliability::BestEffort();
auto reader_qos = subscriber.default_datareader_qos();
reader_qos << Reliability::Reliable(Duration::from_millisecs(200));
dds::pub::DataWriter<Hello::Msg> writer(publisher, topic, writer_qos);
dds::sub::DataReader<Hello::Msg> reader(subscriber, topic, reader_qos);

howto::wait_until([&writer, &reader]() -> bool {
    return writer.offered_incompatible_qos_staus().total_count() > 0
        and reader.requested_incompatible_qos_status().total_count() > 0;
}, "Both endpoints did not report incompatible QoS");
howto::require(writer.publication_matched_status().current_count() == 0, "Incompatible writer matched");

Find the policy behind an incompatible match

Read one RequestedIncompatibleQosStatus snapshot. In this test, the latest policy ID is the ID of Reliability. Print the nonzero counts in policies() and label the known ID.

const auto status = reader.requested_incompatible_qos_status();
const auto reliability_id = dds::core::policy::policy_id<Reliability>::value;
const char* name = "Reliability";
howto::require(status.last_policy_id() == reliability_id, "Expected a Reliability mismatch");
std::cout << "Latest incompatible policy: " << name << '\n';

bool counted_reliability = false;
for (const auto& policy : status.policies())
{
    if (policy.count() == 0)
        continue;

    if (policy.policy_id() == reliability_id)
    {
        std::cout << name << ": " << policy.count() << '\n';
        counted_reliability = true;
    }
    else
    {
        std::cout << "Policy " << policy.policy_id() << ": " << policy.count() << '\n';
    }
}
howto::require(counted_reliability, "Reliability did not appear in the policy counts");

Receive a sample written before the reader joined

Set transient-local durability and reliable delivery on both endpoints. Keep the writer alive so its history retains the sample. A reader created after the write receives it after discovery.

const auto reliable = Reliability::Reliable(Duration::from_millisecs(200));
auto writer_qos = publisher.default_writer_qos();
writer_qos << Durability::TransientLocal() << reliable << History::KeepLast(4);
dds::pub::DataWriter<Hello::Msg> writer(publisher, topic, writer_qos);

char text[] = "before the reader";
Hello::Msg sample{};
sample.userId = 7;
sample.message = text;
writer.write(sample);

auto reader_qos = subscriber.default_datareader_qos();
reader_qos << Durability::TransientLocal() << reliable << History::KeepLast(4);
dds::sub::DataReader<Hello::Msg> reader(subscriber, topic, reader_qos);
howto::wait_for_readers(writer, 1);
howto::wait_until([&reader, &sample]() -> bool {
    const auto samples = reader.take();
    for (const auto& received : samples)
        if (received.info().valid() and received.data().userId == sample.userId)
            return true;

    return false;
}, "The late-joining reader did not receive the retained sample");

Keep the last two samples per key

Set the reader's History::KeepLast(2) before creating it. This example sends three values for one key and uses read() to check that the reader kept the last two. The writer keeps a separate history of three samples.

const auto reliable = Reliability::Reliable(Duration::from_millisecs(200));
auto writer_qos = publisher.default_writer_qos();
writer_qos << reliable << History::KeepLast(3);
auto reader_qos = subscriber.default_datareader_qos();
reader_qos << reliable << History::KeepLast(2);
dds::pub::DataWriter<Hello::Msg> writer(publisher, topic, writer_qos);
dds::sub::DataReader<Hello::Msg> reader(subscriber, topic, reader_qos);
howto::wait_for_readers(writer, 1);

char first[] = "first";
char second[] = "second";
char third[] = "third";
Hello::Msg sample{};
sample.userId = 7;
for (auto* message : {first, second, third})
{
    sample.message = message;
    writer.write(sample);
}

howto::wait_until([&reader, &second, &third]() -> bool {
    const auto samples = reader.read();
    bool received_second = false;
    bool received_third = false;
    for (const auto& received : samples)
    {
        if (not received.info().valid())
            continue;

        received_second = received_second or std::strcmp(received.data().message, second) == 0;
        received_third = received_third or std::strcmp(received.data().message, third) == 0;
    }
    if (received_third)
        howto::require(samples.length() == 2 and received_second, "The reader did not retain the last two samples");

    return received_third;
}, "The last sample did not arrive");

Bound history across two keys

Set History::KeepLast(2) with ResourceLimits(4, 2, 2) on the reader. The arguments limit total samples, instances, and samples per instance.

Send three values for each of two keys. The reader checks that it retained the last two values for each key.

auto reader_qos = subscriber.default_datareader_qos();
reader_qos << reliable << History::KeepLast(2) << ResourceLimits(4, 2, 2);
dds::sub::DataReader<Hello::Msg> reader(subscriber, topic, reader_qos);
howto::wait_for_readers(writer, 1);

char first[] = "first";
char second[] = "second";
char third[] = "third";
for (const auto key : { 7, 8 })
{
    Hello::Msg sample{};
    sample.userId = key;
    for (auto* value : { first, second, third })
    {
        sample.message = value;
        writer.write(sample);
    }
}

howto::wait_until([&reader, &second, &third]() -> bool {
    const auto batch = reader.read();
    if (batch.length() != 4)
        return false;

    std::array<bool, 2> has_second{};
    std::array<bool, 2> has_third{};
    for (const auto& sample : batch)
    {
        howto::require(sample.info().valid(), "Expected a valid sample");
        const auto& data = sample.data();
        howto::require(data.userId == 7 or data.userId == 8, "Unexpected key");
        const auto index = data.userId - 7;
        has_second[index] = has_second[index] or std::strcmp(data.message, second) == 0;
        has_third[index] = has_third[index] or std::strcmp(data.message, third) == 0;
    }
    return has_second[0] and has_second[1] and has_third[0] and has_third[1];
}, "Expected the last two samples for each key");

Choose a writer for each key with exclusive ownership

Enable EXCLUSIVE ownership on both writers and the reader. Give the writers different OwnershipStrength values.

The stronger writer writes key 7 first. The weaker writer then writes key 7 and a separate key 8. The reader checks that the stronger writer supplied key 7 and the weaker writer supplied key 8.

auto weaker_qos = publisher.default_writer_qos();
weaker_qos << reliable << Ownership::Exclusive() << OwnershipStrength(10);
auto stronger_qos = weaker_qos;
stronger_qos << OwnershipStrength(20);
auto reader_qos = subscriber.default_datareader_qos();
reader_qos << reliable << Ownership::Exclusive();
dds::pub::DataWriter<Hello::Msg> weaker(publisher, topic, weaker_qos);
dds::pub::DataWriter<Hello::Msg> stronger(publisher, topic, stronger_qos);
dds::sub::DataReader<Hello::Msg> reader(subscriber, topic, reader_qos);
char strong_value[] = "stronger writer";
char weak_value[] = "weaker writer";
Hello::Msg data{};
data.userId = 7;
data.message = strong_value;
stronger.write(data);
howto::wait_until([&reader, &strong_value]() -> bool {
    const auto batch = reader.take();
    return batch.length() == 1 and batch.begin()->info().valid()
        and batch.begin()->data().userId == 7 and std::strcmp(batch.begin()->data().message, strong_value) == 0;
}, "The stronger writer's key 7 sample did not arrive");

data.message = weak_value;
weaker.write(data);
data.userId = 8;
weaker.write(data);
howto::wait_until([&reader, &weak_value]() -> bool {
    const auto batch = reader.take();
    bool has_key_8 = false;
    for (const auto& sample : batch)
    {
        howto::require(sample.info().valid(), "Expected a valid sample");
        howto::require(sample.data().userId == 8, "The weaker writer updated the owned key");
        has_key_8 = has_key_8 or std::strcmp(sample.data().message, weak_value) == 0;
    }
    return has_key_8;
}, "The weaker writer's independent key 8 sample did not arrive");

QoS delivery behavior

Checks incompatible reliability, receives retained data with transient-local durability, and waits for DDS acknowledgments.

Build the program with the command below. A successful run prints QoS delivery examples passed.

cmake --build . --target example_cpp_howto_delivery
Complete source
#include <cstring>

#include "demo.hpp"

namespace
{
using dds::core::Duration;
using dds::core::policy::Durability;
using dds::core::policy::History;
using dds::core::policy::Reliability;

void report_incompatible_reliability()
{
    dds::domain::DomainParticipant participant(96);
    dds::topic::Topic<Hello::Msg> topic(participant, "HowToIncompatible");
    dds::pub::Publisher publisher(participant);
    dds::sub::Subscriber subscriber(participant);

    // [incompatible_reliability]
    auto writer_qos = publisher.default_writer_qos();
    writer_qos << Reliability::BestEffort();
    auto reader_qos = subscriber.default_datareader_qos();
    reader_qos << Reliability::Reliable(Duration::from_millisecs(200));
    dds::pub::DataWriter<Hello::Msg> writer(publisher, topic, writer_qos);
    dds::sub::DataReader<Hello::Msg> reader(subscriber, topic, reader_qos);

    howto::wait_until([&writer, &reader]() -> bool {
        return writer.offered_incompatible_qos_staus().total_count() > 0
            and reader.requested_incompatible_qos_status().total_count() > 0;
    }, "Both endpoints did not report incompatible QoS");
    howto::require(writer.publication_matched_status().current_count() == 0, "Incompatible writer matched");
    // [incompatible_reliability]

    // [incompatible_qos_counts]
    const auto status = reader.requested_incompatible_qos_status();
    const auto reliability_id = dds::core::policy::policy_id<Reliability>::value;
    const char* name = "Reliability";
    howto::require(status.last_policy_id() == reliability_id, "Expected a Reliability mismatch");
    std::cout << "Latest incompatible policy: " << name << '\n';

    bool counted_reliability = false;
    for (const auto& policy : status.policies())
    {
        if (policy.count() == 0)
            continue;

        if (policy.policy_id() == reliability_id)
        {
            std::cout << name << ": " << policy.count() << '\n';
            counted_reliability = true;
        }
        else
        {
            std::cout << "Policy " << policy.policy_id() << ": " << policy.count() << '\n';
        }
    }
    howto::require(counted_reliability, "Reliability did not appear in the policy counts");
    // [incompatible_qos_counts]
}

void receive_sample_written_before_subscription()
{
    dds::domain::DomainParticipant participant(97);
    dds::topic::Topic<Hello::Msg> topic(participant, "HowToDurability");
    dds::pub::Publisher publisher(participant);
    dds::sub::Subscriber subscriber(participant);

    // [transient_local]
    const auto reliable = Reliability::Reliable(Duration::from_millisecs(200));
    auto writer_qos = publisher.default_writer_qos();
    writer_qos << Durability::TransientLocal() << reliable << History::KeepLast(4);
    dds::pub::DataWriter<Hello::Msg> writer(publisher, topic, writer_qos);

    char text[] = "before the reader";
    Hello::Msg sample{};
    sample.userId = 7;
    sample.message = text;
    writer.write(sample);

    auto reader_qos = subscriber.default_datareader_qos();
    reader_qos << Durability::TransientLocal() << reliable << History::KeepLast(4);
    dds::sub::DataReader<Hello::Msg> reader(subscriber, topic, reader_qos);
    howto::wait_for_readers(writer, 1);
    howto::wait_until([&reader, &sample]() -> bool {
        const auto samples = reader.take();
        for (const auto& received : samples)
            if (received.info().valid() and received.data().userId == sample.userId)
                return true;

        return false;
    }, "The late-joining reader did not receive the retained sample");
    // [transient_local]

    // [wait_for_acknowledgments]
    writer.wait_for_acknowledgments(Duration::from_millisecs(500));
    // [wait_for_acknowledgments]
}

Delivery limits, Publisher acknowledgments, and ownership

Tests per-key retention, Publisher acknowledgment waits, and exclusive ownership across two keys.

Build the program with the command below. A successful run prints Delivery limits and ownership examples passed.

cmake --build . --target example_cpp_howto_delivery_limits
Complete source
#include <array>
#include <cstring>

#include "demo.hpp"

namespace
{
using dds::core::Duration;
using dds::core::policy::History;
using dds::core::policy::Ownership;
using dds::core::policy::OwnershipStrength;
using dds::core::policy::Reliability;
using dds::core::policy::ResourceLimits;

const auto reliable = Reliability::Reliable(Duration::from_millisecs(200));

void keep_two_samples_for_each_key()
{
    dds::domain::DomainParticipant participant(102);
    dds::topic::Topic<Hello::Msg> topic(participant, "HowToHistoryLimits");
    dds::pub::Publisher publisher(participant);
    dds::sub::Subscriber subscriber(participant);
    auto writer_qos = publisher.default_writer_qos();
    writer_qos << reliable << History::KeepLast(3);
    dds::pub::DataWriter<Hello::Msg> writer(publisher, topic, writer_qos);

    // [history_resource_limits]
    auto reader_qos = subscriber.default_datareader_qos();
    reader_qos << reliable << History::KeepLast(2) << ResourceLimits(4, 2, 2);
    dds::sub::DataReader<Hello::Msg> reader(subscriber, topic, reader_qos);
    howto::wait_for_readers(writer, 1);

    char first[] = "first";
    char second[] = "second";
    char third[] = "third";
    for (const auto key : { 7, 8 })
    {
        Hello::Msg sample{};
        sample.userId = key;
        for (auto* value : { first, second, third })
        {
            sample.message = value;
            writer.write(sample);
        }
    }

    howto::wait_until([&reader, &second, &third]() -> bool {
        const auto batch = reader.read();
        if (batch.length() != 4)
            return false;

        std::array<bool, 2> has_second{};
        std::array<bool, 2> has_third{};
        for (const auto& sample : batch)
        {
            howto::require(sample.info().valid(), "Expected a valid sample");
            const auto& data = sample.data();
            howto::require(data.userId == 7 or data.userId == 8, "Unexpected key");
            const auto index = data.userId - 7;
            has_second[index] = has_second[index] or std::strcmp(data.message, second) == 0;
            has_third[index] = has_third[index] or std::strcmp(data.message, third) == 0;
        }
        return has_second[0] and has_second[1] and has_third[0] and has_third[1];
    }, "Expected the last two samples for each key");
    // [history_resource_limits]

    std::cout << "History limits: two keys, two samples each\n";
}

void wait_for_two_writers()
{
    dds::domain::DomainParticipant participant(103);
    dds::topic::Topic<Hello::Msg> first_topic(participant, "HowToPublisherAckFirst");
    dds::topic::Topic<Hello::Msg> second_topic(participant, "HowToPublisherAckSecond");
    dds::pub::Publisher publisher(participant);
    dds::sub::Subscriber subscriber(participant);
    auto writer_qos = publisher.default_writer_qos();
    writer_qos << reliable;
    auto reader_qos = subscriber.default_datareader_qos();
    reader_qos << reliable;
    dds::pub::DataWriter<Hello::Msg> first_writer(publisher, first_topic, writer_qos);
    dds::pub::DataWriter<Hello::Msg> second_writer(publisher, second_topic, writer_qos);
    dds::sub::DataReader<Hello::Msg> first_reader(subscriber, first_topic, reader_qos);
    dds::sub::DataReader<Hello::Msg> second_reader(subscriber, second_topic, reader_qos);

    // [publisher_wait_two_writers]
    howto::wait_for_readers(first_writer, 1);
    howto::wait_for_readers(second_writer, 1);

    char message[] = "publisher acknowledgment";
    Hello::Msg data{};
    data.userId = 7;
    data.message = message;
    first_writer.write(data);
    second_writer.write(data);

    publisher.wait_for_acknowledgments(Duration::from_millisecs(1000));
    // [publisher_wait_two_writers]

    for (auto* reader : { &first_reader, &second_reader })
    {
        howto::wait_until([reader]() -> bool {
            const auto batch = reader->take();
            return batch.length() == 1 and batch.begin()->info().valid() and batch.begin()->data().userId == 7;
        }, "A reader did not receive its sample");
    }
    std::cout << "Publisher acknowledgment: both reliable readers received a sample\n";
}

void choose_an_exclusive_writer()
{
    dds::domain::DomainParticipant participant(104);
    dds::topic::Topic<Hello::Msg> topic(participant, "HowToExclusiveOwnership");
    dds::pub::Publisher publisher(participant);
    dds::sub::Subscriber subscriber(participant);

    // [exclusive_writer_qos]
    auto weaker_qos = publisher.default_writer_qos();
    weaker_qos << reliable << Ownership::Exclusive() << OwnershipStrength(10);
    auto stronger_qos = weaker_qos;
    stronger_qos << OwnershipStrength(20);
    auto reader_qos = subscriber.default_datareader_qos();
    reader_qos << reliable << Ownership::Exclusive();
    dds::pub::DataWriter<Hello::Msg> weaker(publisher, topic, weaker_qos);
    dds::pub::DataWriter<Hello::Msg> stronger(publisher, topic, stronger_qos);
    dds::sub::DataReader<Hello::Msg> reader(subscriber, topic, reader_qos);
    // [exclusive_writer_qos]

    howto::wait_for_readers(weaker, 1);
    howto::wait_for_readers(stronger, 1);

    // [exclusive_writer_keys]
    char strong_value[] = "stronger writer";
    char weak_value[] = "weaker writer";
    Hello::Msg data{};
    data.userId = 7;
    data.message = strong_value;
    stronger.write(data);
    howto::wait_until([&reader, &strong_value]() -> bool {
        const auto batch = reader.take();
        return batch.length() == 1 and batch.begin()->info().valid()
            and batch.begin()->data().userId == 7 and std::strcmp(batch.begin()->data().message, strong_value) == 0;
    }, "The stronger writer's key 7 sample did not arrive");

    data.message = weak_value;
    weaker.write(data);
    data.userId = 8;
    weaker.write(data);
    howto::wait_until([&reader, &weak_value]() -> bool {
        const auto batch = reader.take();
        bool has_key_8 = false;
        for (const auto& sample : batch)
        {
            howto::require(sample.info().valid(), "Expected a valid sample");
            howto::require(sample.data().userId == 8, "The weaker writer updated the owned key");
            has_key_8 = has_key_8 or std::strcmp(sample.data().message, weak_value) == 0;
        }
        return has_key_8;
    }, "The weaker writer's independent key 8 sample did not arrive");
    // [exclusive_writer_keys]

    std::cout << "Exclusive ownership: the stronger writer kept key 7 and the weaker writer supplied key 8\n";
}
}

int main()
{
    return howto::run([]() -> void {
        keep_two_samples_for_each_key();
        wait_for_two_writers();
        choose_an_exclusive_writer();
    }, "Delivery limits and ownership examples passed");
}
